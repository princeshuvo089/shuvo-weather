const cityInput = document.getElementById("cityInput"); const searchBtn = document.getElementById("searchBtn"); const suggestionsList = document.getElementById("suggestionsList"); const cityNameEl = document.getElementById("cityName"); const dateEl = document.getElementById("date"); const weatherCustomIcon = document.getElementById("weatherCustomIcon"); const tempEl = document.getElementById("temp"); const conditionEl = document.getElementById("condition"); const windSpeedEl = document.getElementById("windSpeed"); const humidityEl = document.getElementById("humidity"); const feelsLikeEl = document.getElementById("feelsLike"); const visibilityEl = document.getElementById("visibility"); const hourlyContainer = document.getElementById("hourlyContainer"); const dailyContainer = document.getElementById("dailyContainer"); // উন্নত ও সুন্দর আবহাওয়া আইকন এবং বিবরণ function getWeatherMeta(code, isDay) { if (code === 0) return { desc: isDay ? "পরিষ্কার ও রৌদ্রোজ্জ্বল" : "পরিষ্কার রাত", icon: isDay ? "☀️" : "🌙" }; if (code === 1 || code === 2) return { desc: "আংশিক মেঘলা", icon: isDay ? "🌤️" : "☁️🌙" }; if (code === 3) return { desc: "ঘন মেঘলা", icon: "☁️" }; if (code === 45 || code === 48) return { desc: "কুয়াশাচ্ছন্ন", icon: "🌫️" }; if (code >= 51 && code <= 55) return { desc: "গুঁড়ি গুঁড়ি বৃষ্টি", icon: "🌦️" }; if (code >= 61 && code <= 65) return { desc: "বৃষ্টিপাত", icon: "🌧️" }; if (code >= 80 && code <= 82) return { desc: "ভারী বর্ষণ", icon: "⛈️" }; if (code >= 95) return { desc: "বজ্রঝড়", icon: "⚡" }; return { desc: "স্বাভাবিক আবহাওয়া", icon: isDay ? "🌤️" : "🌙" }; } function getBanglaDate(dateObj = new Date()) { const options = { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' }; return dateObj.toLocaleDateString('bn-BD', options); } // মূল আবহাওয়া ফাংশন (Hourly + 7 Days Forecast) async function fetchWeatherByCoords(lat, lon, displayName) { searchBtn.innerText = "..."; suggestionsList.style.display = "none"; try { const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}¤t=temperature_2m,relative_humidity_2m,apparent_temperature,is_day,weather_code,wind_speed_10m,surface_pressure&hourly=temperature_2m,weather_code,is_day&daily=weather_code,temperature_2m_max,temperature_2m_min&timezone=auto&forecast_days=7`; const res = await fetch(url); const data = await res.json(); const current = data.current; const meta = getWeatherMeta(current.weather_code, current.is_day); cityNameEl.innerText = displayName; dateEl.innerText = getBanglaDate(); tempEl.innerText = Math.round(current.temperature_2m); conditionEl.innerText = meta.desc; weatherCustomIcon.innerText = meta.icon; windSpeedEl.innerText = `${current.wind_speed_10m} km/h`; humidityEl.innerText = `${current.relative_humidity_2m}%`; feelsLikeEl.innerText = `${Math.round(current.apparent_temperature)}°C`; visibilityEl.innerText = `${Math.round(current.surface_pressure)} hPa`; // ২৪ ঘণ্টার পূর্বাভাস তৈরি hourlyContainer.innerHTML = ""; const currentHourIndex = new Date().getHours(); for (let i = currentHourIndex; i < currentHourIndex + 24 && i < data.hourly.time.length; i++) { const timeStr = new Date(data.hourly.time[i]).toLocaleTimeString('bn-BD', { hour: 'numeric', hour12: true }); const hMeta = getWeatherMeta(data.hourly.weather_code[i], data.hourly.is_day[i]); const hTemp = Math.round(data.hourly.temperature_2m[i]); hourlyContainer.innerHTML += `
${timeStr} ${hMeta.icon} ${hTemp}°C
`; } // ৭ দিনের পূর্বাভাস তৈরি dailyContainer.innerHTML = ""; for (let j = 0; j < data.daily.time.length; j++) { const dayDate = new Date(data.daily.time[j]); let dayName = j === 0 ? "আজ" : j === 1 ? "আগামীকাল" : dayDate.toLocaleDateString('bn-BD', { weekday: 'long' }); const dMeta = getWeatherMeta(data.daily.weather_code[j], 1); const maxT = Math.round(data.daily.temperature_2m_max[j]); const minT = Math.round(data.daily.temperature_2m_min[j]); dailyContainer.innerHTML += `
${dayName}
${dMeta.icon} ${dMeta.desc}
${maxT}° ${minT}°
`; } } catch (err) { alert("উপাত্ত লোড করতে সমস্যা হয়েছে!"); } finally { searchBtn.innerText = "অনুসন্ধান"; } } // সার্চ ড্রপডাউন ও অটো-সাজেশন লজিক let debounceTimer; cityInput.addEventListener("input", () => { clearTimeout(debounceTimer); const query = cityInput.value.trim(); if (query.length < 2) { suggestionsList.style.display = "none"; return; } debounceTimer = setTimeout(async () => { try { const geoUrl = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(query)}&count=6&language=en&format=json`; const res = await fetch(geoUrl); const geoData = await res.json(); if (!geoData.results || geoData.results.length === 0) { suggestionsList.style.display = "none"; return; } suggestionsList.innerHTML = ""; geoData.results.forEach((loc) => { const li = document.createElement("li"); li.className = "suggestion-item"; const district = loc.admin1 ? `${loc.admin1}, ` : ""; li.innerHTML = ` 📍 ${loc.name} ${district}${loc.country_code ? loc.country_code.toUpperCase() : 'BD'} `; li.addEventListener("click", () => { cityInput.value = loc.name; suggestionsList.style.display = "none"; fetchWeatherByCoords(loc.latitude, loc.longitude, `${loc.name}, ${loc.country_code ? loc.country_code.toUpperCase() : 'BD'}`); }); suggestionsList.appendChild(li); }); suggestionsList.style.display = "block"; } catch (e) { suggestionsList.style.display = "none"; } }, 250); }); // সরাসরি সার্চ বাটনে চাপ দিলে searchBtn.addEventListener("click", async () => { const query = cityInput.value.trim(); if (!query) return; const geoUrl = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(query)}&count=1&language=en&format=json`; const res = await fetch(geoUrl); const geoData = await res.json(); if (geoData.results && geoData.results.length > 0) { const loc = geoData.results[0]; fetchWeatherByCoords(loc.latitude, loc.longitude, `${loc.name}, ${loc.country_code ? loc.country_code.toUpperCase() : 'BD'}`); } }); // কীবোর্ডের Enter বাটনে চাপ দিলে cityInput.addEventListener("keypress", (e) => { if (e.key === "Enter") searchBtn.click(); }); // স্ক্রিনের অন্য কোথাও ক্লিক করলে ড্রপডাউন বন্ধ হওয়া document.addEventListener("click", (e) => { if (!e.target.closest(".search-wrapper")) { suggestionsList.style.display = "none"; } }); // প্রথমবার ওপেন হলে কুড়িগ্রাম লোড হবে window.addEventListener("DOMContentLoaded", () => { fetchWeatherByCoords(25.8054, 89.6362, "Kurigram, BD"); });
